# Lumen Drift: one CMake project, three builds.
#   PC:                 cmake -B build && cmake --build build                (SDL2 + OpenGL, the original game)
#   PC Vita simulation: vita sim  (= -DVITA_SIM=ON: 960x544, Vita controls, memory cap; cmake/VitaPort.cmake)
#   PS Vita:            vita build (VitaSDK toolchain: vitaGL + vitaport + SDL2 audio -> eboot.bin, lumen-drift.vpk)
cmake_minimum_required(VERSION 3.20)
project(lumen-drift LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

include(cmake/VitaPort.cmake)

add_executable(lumen-drift src/main.cpp src/game.cpp src/render.cpp src/audio.cpp)
target_include_directories(lumen-drift PRIVATE src)
if(MSVC)
  target_compile_options(lumen-drift PRIVATE /W4 /permissive- /utf-8)
  target_compile_definitions(lumen-drift PRIVATE _CRT_SECURE_NO_WARNINGS SDL_MAIN_HANDLED)
else()
  target_compile_options(lumen-drift PRIVATE -Wall -Wextra)
endif()

if(VITA)
  # SDL2 is used for audio only; the vdpm package (sdl2 or sdl2_vitagl) brings its Sce stubs through SDL2::SDL2.
  find_package(SDL2 CONFIG REQUIRED)
  target_sources(lumen-drift PRIVATE src/platform/platform_vita.cpp)
  vitaport_link_vita(lumen-drift)
  target_link_libraries(lumen-drift PRIVATE ${VITAPORT_VITAGL_LIBS} SDL2::SDL2)
  vitaport_package(lumen-drift)
else()
  option(LUMEN_FETCH_SDL "Download and build SDL2 when it isn't installed" ON)
  find_package(SDL2 CONFIG QUIET)
  if(NOT TARGET SDL2::SDL2 AND NOT TARGET SDL2::SDL2-static AND LUMEN_FETCH_SDL)
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL2 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-2.32.8 GIT_SHALLOW TRUE)
    FetchContent_MakeAvailable(SDL2)
  endif()
  find_package(OpenGL REQUIRED)
  target_sources(lumen-drift PRIVATE src/platform/platform_sdl2.cpp)
  if(TARGET SDL2::SDL2main)
    target_link_libraries(lumen-drift PRIVATE SDL2::SDL2main)
  endif()
  if(TARGET SDL2::SDL2-static)
    target_link_libraries(lumen-drift PRIVATE SDL2::SDL2-static)
  else()
    target_link_libraries(lumen-drift PRIVATE SDL2::SDL2)
  endif()
  target_link_libraries(lumen-drift PRIVATE OpenGL::GL)
  if(VITA_SIM)
    vitaport_link_sim(lumen-drift)
  endif()
  # one predictable output folder for every generator: <build>/bin/lumen-drift[.exe]
  set_target_properties(lumen-drift PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin/$<0:>")
endif()
